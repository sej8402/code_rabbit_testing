@PostMapping("/upload-feed-file")
	public ResponseEntity<String> uploadFeedFile(
			@RequestParam("channel") String platform) {
		File uploadedFile = null;
		File processedFileRef = null;

		try {

			String processedFilePath = xpservice.processFilesFromStaticPath2(platform, uploadedFile);
			processedFileRef = new File(processedFilePath);

			return ResponseEntity.ok(processedFileRef.getName());

		} catch (Exception e) {
			e.printStackTrace();
			return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
					.body("File upload failed: " + e.getMessage());

		} finally {
			// Optional: Cleanup uploaded file
			if (uploadedFile != null && uploadedFile.exists()) {
				try {
					Files.delete(uploadedFile.toPath());
				} catch (IOException e) {
					System.err.println("Failed to delete uploaded file: " + e.getMessage());
				}
			}
		}
	}
