        // New function to handle feed processing
        function processFeed() {
            const url = document.getElementById('client_product_url').value;
            const fileInput = document.getElementById('feedFile');
            const file = fileInput.files[0];

            if (!url && !file) {
                showTooltip('Please provide either a product URL or upload a feed file');
                return;
            }

            // Hide all report tabs and content
            document.querySelector('.report-tabs').style.display = 'none';
            document.querySelector('.report-content').style.display = 'none';

            // Show loading state
            const rightPane = document.querySelector('.right-pane');
            const loadingPlaceholder = document.createElement('div');
            loadingPlaceholder.className = 'no-data-placeholder';
            loadingPlaceholder.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Analyzing feed, please wait...';
            rightPane.appendChild(loadingPlaceholder);

            // Simulate processing time
            setTimeout(() => {
                loadingPlaceholder.remove();

                // Hide the report section and show the missing attributes form
                document.querySelector('.right-pane-header').style.display = 'none';
                document.getElementById('missing-attributes-container').style.display = 'block';

                showTooltip('Feed analysis completed!');
            }, 2000);
        }

        // Handle file selection display
        document.getElementById('feedFile').addEventListener('change', function (e) {
            const fileName = e.target.files[0] ? e.target.files[0].name : 'No file chosen';
            document.getElementById('fileName').textContent = fileName;
        });

        // Handle image uploads
        function handleImageUpload(event, type) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function (e) {
                const imageUrl = e.target.result;

                if (type === 'main') {
                    document.getElementById('uploadPlaceholder').style.display = 'none';
                    document.getElementById('mainImageContainer').style.backgroundImage = `url(${imageUrl})`;
                    document.getElementById('mainImageContainer').style.backgroundSize = 'cover';
                } else {
                    const thumbIndex = type.replace('thumb', '');
                    const thumbs = document.querySelectorAll('.thumbnail');
                    thumbs[thumbIndex - 1].innerHTML = '';
                    thumbs[thumbIndex - 1].style.backgroundImage = `url(${imageUrl})`;
                    thumbs[thumbIndex - 1].style.backgroundSize = 'cover';
                }
            }
            reader.readAsDataURL(file);
        }

        // Modified toggleForm function with proper state restoration
        function toggleForm(type) {
            // Hide all form sections
            document.querySelectorAll('.form-section').forEach(form => {
                form.classList.remove('active');
            });
            document.getElementById(type + '-form').classList.add('active');

            // Get references to UI elements
            const reportTabs = document.querySelector('.report-tabs');
            const reportContents = document.querySelectorAll('.report-content');
            const rightPaneHeader = document.querySelector('.right-pane-header');
            const missingAttributes = document.getElementById('missing-attributes-container');

            if (type === 'report') {
                // Feed Transformation: Hide reports UI
                reportTabs.style.display = 'none';
                reportContents.forEach(content => {
                    content.style.display = 'none';
                });
                rightPaneHeader.style.display = 'none';
                missingAttributes.style.display = 'none';
            } else {
                // Competitive Report: Show reports UI
                reportTabs.style.display = 'flex';
                rightPaneHeader.style.display = 'flex';

                // Remove inline display styles to allow CSS classes to work
                reportContents.forEach(content => {
                    content.style.display = '';
                });

                // Reinitialize charts if needed
                if (document.querySelector('.report-content.active').id === 'summary-report') {
                    initSummaryCharts();
                }
            }
        }

        // Add this to DOMContentLoaded to initialize correctly

        function Feed_Transformation() {

            const loader = document.querySelector('.loader-container');
            const rightPane = document.querySelector('.right-pane');

            loader.style.display = 'flex';

                return;
            }

            var fdata = {
                "urls": urlInput,
                "channel": selectedChannel
            };

            $.ajax({
                url: APICallingLink + getCrawledData,
                type: "POST",
                data: $.param(fdata),
                contentType: 'application/x-www-form-urlencoded',
                async: true,
                success: function (filename) {
                    console.log("Upload response filename:", filename);
                    if (!filename) {
                        alert("No file returned.");
                        return;

                    document.body.removeChild(link);
                    loader.style.display = 'none';
                },
                error: function (data) {
                    loader.style.display = 'none';
                    alert("Error :: " + JSON.stringify(data));
                }
            });
        }

        //Added by yugant 10-06-2025 for validating and fetching product data
        function validateAndFetchData() {
            let isValid = true;

            // Validation: Check required fields
            $(".attribute-row").each(function () {
                const errorMessage = $(this).find(".error-message");
                const inputField = $(this).find(".attribute-value");

                if (errorMessage.is(":visible") && inputField.val().trim() === "") {
                    inputField.css("border", "1px solid red");
                    console.warn("Mandatory field missing for:", $(this).data("key"));
                    isValid = false;
                } else {
                    inputField.css("border", "");
                }
            });

            if (!isValid) {
                console.warn("Mandatory fields missing. Aborting API call.");
                return false;
            }

            // Validation: Check max-length
            let anyExceeded = false;
            $(".attribute-row").each(function () {
                const $row = $(this);
                const $input = $row.find(".attribute-value");
                const maxLen = parseInt($row.data("max-length") || 100);
                const currentLen = $input.val().length;

                if (currentLen > maxLen) {
                    $input.css("border", "1px solid red");
                    $row.attr("data-exceeded", "true");
                     const keyName = $row.data("key-check") || $row.data("key") || "Unknown";
                $("#mandatory-status-text").text(`Length Exceeded – ${keyName}: ${currentLen} / ${maxLen} chars`);
    
                    console.warn(`Length exceeded for key=${$row.data("key")}, Value=${$input.val()}, Max=${maxLen}`);
                    anyExceeded = true;
                } else {
                    $input.css("border", "");
                    $row.removeAttr("data-exceeded");
                }
            });

            if (anyExceeded) {
                 $("#mandatory-status-text").text(""); 
                console.warn("One or more fields exceeded max length.");
                return;
            }

            // Fetch data from backend and populate
            $.ajax({
                url: APICallingLink + findProductData,
                type: "GET",
                success: function (response) {
                    console.log(" Raw response from API:", response);

                    $(".attribute-row").each(function () {
                        const rawKey = $(this).data("key");
                        const normalizedKey = normalizeAttributeName(rawKey);
                        console.log("Searching for Key:", rawKey, "  Normalized:", normalizedKey);

                        let matchedValue = "";   //comparing with mongo key 

                        for (const mongoKey in response) {
                            const normalizedMongoKey = normalizeAttributeName(mongoKey);
                            // console.log("Comparing with Mongo Key:", mongoKey, " ", normalizedMongoKey);

                            if (normalizedMongoKey === normalizedKey) {
                                matchedValue = response[mongoKey];
                                //console.log("Match found:", mongoKey, "-", matchedValue);
                                break;
                            }
                        }

                        if (matchedValue) {
                            $(this).find(".attribute-value").val(matchedValue);
                        } else {
                            console.warn("No matching key found in API response for:", rawKey);
                        }
                    });
                },
                error: function (err) {
                    console.error("Failed to fetch data from API", err);
                }
            });

            return true;
        }

        function normalizeAttributeName(name) {

        $(".btn-save").click(function () {
            const selectedOption = $('#Channel_option option:selected');
            const projectId = selectedOption.data('project-id');
            let currentData = {}; // Ideally filled during validateAndFetchData if needed


            const selectedChannel = document.getElementById("Channel_option").value;
            // Proceed to fetch existing mappings
             $("#loader").show();
            $.ajax({
                url: APICallingLink + getMappedAttributes + "?channel=" + selectedChannel,
                method: 'GET',
                success: function (res) {
                    const existingMappings = Array.isArray(res)
                        ? res
                        : Array.isArray(res.mappings)
                            ? res.mappings
                            : [];

                    const propertyList = [];
                    const mappingList = [];

                    $(".attribute-row").each(function () {
                        const key = $(this).data("key");
                        const inputVal = $(this).find(".attribute-value").val()?.trim();
                        const selectedAttrName = $(this).find(".map-select option:selected").text()?.trim();

                        // Add property only if not already in currentData
                        if (!currentData[key] && inputVal) {
                            const propertyMap = {};
                            propertyMap[key] = inputVal;
                            propertyList.push(propertyMap);
                        }

                        if (selectedAttrName && selectedAttrName !== "-- Select Attribute --") {
                            const existing = existingMappings.find(item =>
                                normalizeAttributeName(item.sourceAttribute) === normalizeAttributeName(key)
                            );

                            const currentTarget = existing ? normalizeAttributeName(existing.targetAttribute) : null;
                            const selectedNormalized = normalizeAttributeName(selectedAttrName);

                            if (existing && currentTarget !== selectedNormalized) {
                                mappingList.push({
                                    sourceAttribute: key,
                                    targetAttribute: selectedAttrName
                                });
                            }

                            if (!existing) {
                                mappingList.push({
                                    sourceAttribute: key,
                                    targetAttribute: selectedAttrName
                                });
                            }
                        }
                    });

                    if (propertyList.length === 0 && mappingList.length === 0) {
                        $("#loader").hide();
                        alert("No new data to save for this skew.");
                        return;
                    }

                    const payload = {
                        properties: propertyList,
                        mappings: mappingList,
                        count: count,
                        channel: selectedChannel
                    };

                    $.ajax({
                        url: APICallingLink + saveAttributes,
                        type: "POST",
                        contentType: "application/json",
                        data: JSON.stringify(payload),
                        success: function () {
                             $("#loader").hide(); 
                            alert("Data saved for current product.");
                        },
                        error: function (err) {
                              $("#loader").hide();
                            console.error("Error saving data", err);
                            alert("Failed to save data.");
                        }
                    });
                },
                error: function () {
                     $("#loader").hide(); 
                    alert("Failed to fetch existing mapped attributes.");
                }
            });
        });

        function attachCharacterCountListener() {
    $(".attribute-value").off("input").on("input", function () {
        const $input = $(this);
        const $row = $input.closest('.attribute-row');
        const maxLen = parseInt($row.data("max-length") || 100);
        const currentLen = $input.val().length;

        let $counter = $row.find(".char-counter");
        if ($counter.length === 0) {
            $counter = $('<div class="char-counter" style="font-size: 11px; color: grey; margin-top: 2px;"></div>');
            $row.append($counter);
        }

        $counter.text(`${currentLen} / ${maxLen} chars`).show();

        if (currentLen > maxLen) {
            console.log("Length Exceeded triggered", $row.data("key-check"), currentLen, "/", maxLen);
            $input.css("border", "1px solid red");
            $counter.css("color", "red");
            $row.attr("data-exceeded", "true");
            $("#mandatory-status-text").text(`Length Exceeded – ${$row.data("key-check")}: ${currentLen} / ${maxLen} chars`);
        } else {
            $input.css("border", "");
            $counter.css("color", "grey");
            $row.removeAttr("data-exceeded");

            if ($(".attribute-row[data-exceeded='true']").length === 0) {
                $("#mandatory-status-text").text("Mandatory Fields");
            }
        }
    });
}

        function loadAttributesByProjectId() {
            const selectedOption = $('#Channel_option option:selected');
            const projectIdString = selectedOption.data('project-id');

            if (!projectIdString) {
                alert("Please select a valid channel.");
                return;
            }

            // Support single or comma-separated project IDs
            const projectIds = projectIdString.includes(',')
                ? projectIdString.split(',').map(id => id.trim())
                : [projectIdString.trim()];

            const queryParams = projectIds.map(id => `projectIds=${encodeURIComponent(id)}`).join('&');


            // Helper to normalize attribute names
            function normalize(name) {
                return name
                    ?.trim()
                    .replace(/&/g, 'and')       // & with 'and'
                    .replace(/\./g, '@@@')      // . with '@@@'
                    .replace(/\s+/g, '_')       // spaces with underscores
                    .toUpperCase();
            }


            // Get attributes from DB
            $.ajax({
                url: `${APICallingLink + getAttributes}?${queryParams}`,
                method: 'GET',
                success: function (attributes) {
                    // Step 2: Get mapped attributes from Excel
                    const selectedChannel = document.getElementById("Channel_option").value;
                    const channelAttrKey = selectedChannel.toLowerCase() + "Attribute";  //added by sejal 

                    $.ajax({
                        url: APICallingLink + getMappedAttributes + "?channel=" + selectedChannel,
                        method: 'GET',
                        success: function (mappedData) {

                            console.log("Row count:", $(".attribute-row").length);
                            $(".attribute-row").each(function () {
                                console.log("Dropdown found:", $(this).find(".map-select").length);
                            });

                            $(".attribute-row").each(function () {
                                const $row = $(this);
                                const $select = $row.find(".map-select");
                                const clientKey = $row.data("key");

                                if ($select.hasClass("select2-hidden-accessible")) {
                                    $select.select2('destroy');
                                }

                                $select.empty().append('<option value="">-- Select Attribute --</option>');

                                const mapped = mappedData.find(mapping =>
                                    normalize(mapping.sourceAttr) === normalize(clientKey)
                                );



                                attributes.forEach(attr => {
                                    const normalizedAttrName = normalize(attr.attributeName);
                                    const expectedMappedName = mapped ? normalize(mapped.targetAttr) : "";

                                    const isSelected = normalizedAttrName === expectedMappedName;

                                    $select.append(
                                        `<option value="${attr.attributeId}" ${isSelected ? 'selected' : ''}>${attr.attributeName}</option>`
                                    );
                                });

                                // $select.select2({
                                //     placeholder: "-- Select Attribute --",
                                //     allowClear: true,
                                //     width: '100%',
                                //     theme: 'default'
                                // });
                            });
                        },
                        error: function () {
                            alert("Failed to fetch mapped attributes.");
                        }
                    });
                },
                error: function () {
                    alert("Failed to fetch attributes.");
                }
            });
        }

        // Added by Rutuja on 25/06/2025 to extract the brand name from the URL
        function extractBrandNameParts(url) {
            try {
                const parsedUrl = new URL(url);
                const hostname = parsedUrl.hostname.replace(/^www\./, "");
                const domainPlatform = hostname.split('.')[0];
                const platform = domainPlatform.charAt(0).toUpperCase() + domainPlatform.slice(1).toLowerCase();
                const pathSegments = parsedUrl.pathname.split('/').filter(Boolean);
                let brand = domainPlatform; // fallback

                if (pathSegments.length > 0) {
                    const possibleBrand = pathSegments[0].split('-')[0];
                    if (possibleBrand && possibleBrand.length > 2) {
                        brand = possibleBrand;
                    }
                }
                brand = brand.charAt(0).toUpperCase() + brand.slice(1).toLowerCase();
                return { platform, brand };
            } catch (e) {
                return { platform: "Unknown", brand: "Unknown" };
            }
        }
